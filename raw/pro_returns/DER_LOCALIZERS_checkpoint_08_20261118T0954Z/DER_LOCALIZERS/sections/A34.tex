\subsection{The two remaining stable-part questions}
\begin{claim}{A3.1}{Proved}{The zero stable part and a nonzero part without compact objects}\label{A31}
If a reversible fundamental localizer has $T(W)=0$, then $W=W_{\Z}$. For every prime $p$ there is an accessible reversible fundamental localizer $W_p$ with
\[
 T(W_p)=L_p=\loc\bigl(\Sigma^\infty K(\F_p,2)\bigr)\ne0,
 \qquad L_p\cap\Ho(\Sp)^\omega=0.
\]
This construction by itself does not decide whether $W_p=RT(W_p)$.
\end{claim}
\begin{proof}
A.4 gives $W_{\Z}\subseteq W$. If $T(W)=0$, the suspension spectrum of the pointed cofiber of each $W$-map is zero. Its reduced integral homology is zero, so that map is an integral homology equivalence. This gives the reverse inclusion.

Put $K=K(\F_p,2)$ and let $\mathcal A_p$ be the least pointed class containing $K$ and the point, closed under weak equivalences, pointed homotopy colimits, retracts and the three cofiber implications. Define $W_p$ by $C(Bf)\in\mathcal A_p$. The proof of A.7 gives all fundamental and reverse axioms and accessibility, since $K$ has a countable CW model. Suspension spectra preserve the indicated constructions, so $T(W_p)\subseteq L_p$. Conversely $K\to *$ is in $W_p$, and its cofiber spectrum is $\Sigma\Sigma^\infty K$, so equality holds. Its degree-two reduced homology is $\F_p$, proving $L_p\ne0$.

For the assertion about finite spectra we use \cite[Corollary 1.4.1]{AAllenYuan}: the augmentation $\Sigma^\infty K_+\to\mathbb S$ induces an equivalence on sphere-valued function spectra for any simply connected finite homotopy type in that theorem's sense (a $\pi$-finite space). Thus the reduced dual $F(\Sigma^\infty K,\mathbb S)$ is zero. The class of spectra with zero sphere-valued function spectrum is localizing: the contravariant function functor converts cofibers to fibers and coproducts to products. It therefore contains $L_p$. For a finite spectrum its sphere-valued dual vanishes only if the spectrum itself vanishes, by finite duality. This proves the compact-intersection assertion. It rechecks the exact input in REVLOC D.2.6 without assuming a compact generator of $L_p$.
\end{proof}

\begin{claim}{A4.1}{Proved}{The established bounds in the primary interval}\label{A41}
Let $W_{\mathrm{rev}}$ be the least reversible fundamental localizer containing $BC_2\to e$, and put $L=\loc(\mathbb S/2)$. Then
\[
 W_{\mathrm{rev}}\ \subseteq\ W_*:=W_S\cap W_{\Z[1/2]}
 \ \subsetneq\ W_{\Z[1/2]}=R(L),
 \qquad T(W_{\mathrm{rev}})=T(W_*)=L.
\]
The proof here does not make the first inclusion strict. The named functor $P\to e$ of A.5 lies outside $W_*$ and inside its displayed upper endpoint.
\end{claim}
\begin{proof}
Intersections preserve the fundamental and reverse axioms. Both classes in the intersection contain $BC_2\to e$, so minimality gives the first inclusion. The second is strict by A.5, since the finite space $BP\simeq\RP^2$ has zero reduced $\Z[1/2]$-homology. A.6 gives $L\subseteq T(W_{\mathrm{rev}})$, while the upper endpoint gives the opposite inclusion for both displayed stable parts. Finally a cofiber spectrum of a category map is connective. Its localization away from $2$ vanishes exactly when its $\Z[1/2]$-homology vanishes, by connective stable Hurewicz, proving $R(L)=W_{\Z[1/2]}$.
\end{proof}
